local function updatePlayerList()
    local playersList = {}
    for _, player in pairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            table.insert(playersList, player.Name)
        end
    end
    if #playersList == 0 then
        table.insert(playersList, "Nenhum Jogador No Servidor")
    end
    return playersList
end

local function updateCamera()

end

local function headsitOnPlayer(targetPlayer)
    if not targetPlayer or not targetPlayer.Character then return end
    local char = LocalPlayer.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    local tHead = targetPlayer.Character:FindFirstChild("Head")
    if root and tHead then
        root.CFrame = tHead.CFrame * CFrame.new(0, 2.2, 0)
        local weld = Instance.new("WeldConstraint", root)
        weld.Part0 = root
        weld.Part1 = tHead
        char.Humanoid.Sit = true
    end
end

local function removeHeadsit()
    if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then
        for _, v in pairs(LocalPlayer.Character.HumanoidRootPart:GetChildren()) do
            if v:IsA("WeldConstraint") then v:Destroy() end
        end
        LocalPlayer.Character.Humanoid.Sit = false
    end
end

local function EquiparSofaGarantido()
    local character = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
    local humanoid = character:WaitForChild("Humanoid", 3)
    local backpack = LocalPlayer:WaitForChild("Backpack", 3)
    if not humanoid or not backpack then return false end

    if character:FindFirstChild("Couch") then return true end

    if not backpack:FindFirstChild("Couch") then
        pcall(function()
            local toolRemote = ReplicatedStorage:FindFirstChild("RE") and ReplicatedStorage.RE:FindFirstChild("1Too1l")
            if toolRemote then toolRemote:InvokeServer("PickingTools", "Couch") end
        end)
        local t0 = tick()
        repeat task.wait(0.01) until backpack:FindFirstChild("Couch") or (tick() - t0 > 1.5)
    end

    local couchTool = backpack:FindFirstChild("Couch")
    if couchTool then
        humanoid:EquipTool(couchTool)
        task.wait(0.01)
        return true
    end
    return character:FindFirstChild("Couch") ~= nil
end

local function ExecutarKillCouch(targetPlayer)

    local myHRP = LocalPlayer.Character:FindFirstChild("HumanoidRootPart")

    local originalCFrame = myHRP.CFrame

    local runningLoop = true

    local killConn

    killConn = RunService.Heartbeat:Connect(function()
        if not runningLoop or not tHRP or not myHRP then

            return

            myHRP.CFrame = tHRP.CFrame * CFrame.Angles(math.rad(Workspace.DistributedGameTime * 5000), 0, 0)

                myHRP.CFrame = CFrame.new(Vector3.new(9, -67, 315))

end

local function flingWithBus(targetPlayer)
    if not targetPlayer or not targetPlayer.Character or not LocalPlayer.Character then return end
    local myHRP = LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    local humanoid = LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
    if not myHRP or not humanoid then return end

    busSavedPos = myHRP.Position

    pcall(function()
        myHRP.CFrame = CFrame.new(Vector3.new(1181.83, 76.08, -1158.83))
    end)
    task.wait(0.3)

    pcall(function()
        ReplicatedStorage.RE["1Ca1r"]:FireServer("DeleteAllVehicles")
        ReplicatedStorage.RE["1Ca1r"]:FireServer("PickingCar", "SchoolBus")
    end)
    task.wait(0.8)

    local vehicles = Workspace:FindFirstChild("Vehicles")
    local bus = vehicles and vehicles:FindFirstChild(LocalPlayer.Name .. "Car")
    if not bus then return end

    pcall(function()
        myHRP.CFrame = CFrame.new(Vector3.new(1171.15, 79.45, -1166.2))
        humanoid:ChangeState(Enum.HumanoidStateType.Seated)
    end)
    task.wait(0.5)

    for _, part in ipairs(bus:GetDescendants()) do
        if part:IsA("BasePart") then part.CanCollide = false end
    end

    busRunning = true
    local startTime = tick()
    busFlingConn = RunService.Heartbeat:Connect(function()
        if not busRunning or not targetPlayer.Character or not targetPlayer.Character:FindFirstChild("HumanoidRootPart") then
            StopTestFling()
            return
        end

        local tHRP = targetPlayer.Character.HumanoidRootPart
        pcall(function()
            bus:PivotTo(tHRP.CFrame * CFrame.Angles(math.rad(tick() * 1000), math.rad(tick() * 1200), 0))
        end)

        if tick() - startTime > 8 then
            StopTestFling()
        end
    end)
end

TabTroll:AddSection({"Seleção de Jogadores"})

local PlayerDrop = TabTroll:AddDropdown({
    Name = "Lista de Jogadores",
    Options = updatePlayerList(),
    Default = "Selecione...",
    Callback = function(v) _G.SelectedPlayer = v end
})

TabTroll:AddButton({
    Name = "Atualizar Lista de Jogadores", 
    Callback = function() PlayerDrop:Set(updatePlayerList()) end
})

TabTroll:AddToggle({
    Name = "Visualizar Jogador (View)",
    Default = false,
    Callback = function(s) viewEnabled = s updateCamera() end
})

TabTroll:AddButton({
    Name = "Teleportar para o Jogador",
    Callback = function()
        local T = Players:FindFirstChild(_G.SelectedPlayer)
        if T and T.Character and LocalPlayer.Character then
            LocalPlayer.Character.HumanoidRootPart.CFrame = T.Character.HumanoidRootPart.CFrame
        end
    end
})

TabTroll:AddToggle({
    Name = "Headsit (Sentar na Cabeça)",
    Default = false,
    Callback = function(s)
        local T = Players:FindFirstChild(_G.SelectedPlayer)
        if s and T then headsitOnPlayer(T) else removeHeadsit() end
    end
})

TabTroll:AddSection({"Ataques com Sofá"})

TabTroll:AddButton({
    Name = "Kill Couch (1x)",
    Callback = function()
        local T = Players:FindFirstChild(_G.SelectedPlayer)
        if T then ExecutarKillCouch(T) end
    end
})

TabTroll:AddToggle({
    Name = "Auto Kill Couch",
    Default = false,
    Callback = function(state)
        _G.AutoKillCouch = state
        if state then
            task.spawn(function()
                while _G.AutoKillCouch do
                    local T = Players:FindFirstChild(_G.SelectedPlayer)
                    if T then ExecutarKillCouch(T) end
                    task.wait(1)
                end
            end)
        end
    end
})

TabTroll:AddButton({
    Name = "Bring Couch (1x)",
    Callback = function()
        local T = Players:FindFirstChild(_G.SelectedPlayer)
        if T then ExecutarBringCouch(T) end
    end
})

TabTroll:AddButton({
    Name = "Fling Couch (1x)",
    Callback = function()
        local T = Players:FindFirstChild(_G.SelectedPlayer)
        if T then ExecutarFlingCouch(T) end
    end
})

TabTroll:AddSection({"Black Hole"})

TabTroll:AddToggle({
    Name = "Black Hole",
    Default = false,
    Callback = function(state)
        _G_BH_Active = state
        if state then
            SetupBlackHole()
            task.spawn(function()
                while _G_BH_Active do
                    local T = Players:FindFirstChild(_G.SelectedPlayer)
                    if T and T.Character and T.Character:FindFirstChild("HumanoidRootPart") then
                        BH_Part.CFrame = T.Character.HumanoidRootPart.CFrame
                        for _, v in ipairs(Workspace:GetDescendants()) do ForcePart(v) end
                    end
                    RunService.Heartbeat:Wait()
                end
            end)
        end
    end
})

TabTroll:AddSection({"Fling Ônibus"})

TabTroll:AddToggle({
    Name = "Iniciar Fling Ônibus",
    Default = false,
    Callback = function(state)
        if state then
            local target = Players:FindFirstChild(_G.SelectedPlayer)
            if target then flingWithBus(target) end
        else
            StopTestFling()
        end
    end
})
