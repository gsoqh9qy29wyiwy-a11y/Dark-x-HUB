-- Funções Auxiliares
local function SendNotification(title, text, duration)
    pcall(function()
        Rayfield:Notify({
            Title = title,
            Content = text,
            Duration = duration or 3,
            Image = 4483362458
        })
    end)
end

local function ObterListaNomes()
    local lista = {"...", "Todos"}
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            table.insert(lista, player.Name)
        end
    end
    return lista
end

local function AtualizarCacheProps()
    local workspaceCom = workspace:FindFirstChild("WorkspaceCom")
    if not workspaceCom then return end
    
    local novosProps = {}
    for _, cat in ipairs(workspaceCom:GetChildren()) do
        for _, item in ipairs(cat:GetChildren()) do
            if item.Name == MeuPropNome and item:FindFirstChild("SetCurrentCFrame") then
                table.insert(novosProps, item)
            end
        end
    end
    meusPropsCache = novosProps
end

local function AlgumModoAtivo()
    return (PropMovementMode ~= "Nenhum" or PropsActive or ModoAvancadoAtivo)
end

-- UI: SEÇÃO DE JOGADORES
PropsTab:CreateSection("Lista de Jogadores")

local DropdownJogadores
DropdownJogadores = PropsTab:CreateDropdown({
    Name = "Selecionar Jogador",
    Options = ObterListaNomes(),
    CurrentOption = {"..."},
    MultipleOptions = false,
    -- Removido o Flag para evitar conflito com o salvamento de configurações em disco
    Callback = function(opcoes)
        local nomeSelecionado = type(opcoes) == "table" and opcoes[1] or opcoes
        
        -- Ignora se vier nulo ou vazio (evita desmarcar ao recarregar a lista)
        if not nomeSelecionado or nomeSelecionado == "" then
            return
        end

        if nomeSelecionado == "..." then
            jogadorSelecionado = nil
            selecionarTodos = false
        elseif nomeSelecionado == "Todos" then
            selecionarTodos = true
            jogadorSelecionado = nil
        else
            selecionarTodos = false
            local alvo = Players:FindFirstChild(nomeSelecionado)
            if alvo then
                jogadorSelecionado = alvo
            end
        end
        
        if espionando and jogadorSelecionado and jogadorSelecionado.Character then  
            local hum = jogadorSelecionado.Character:FindFirstChildWhichIsA("Humanoid")
            if hum then Camera.CameraSubject = hum end  
        end  
    end
})

local function AtualizarListaDropdown()
    local novasOpcoes = ObterListaNomes()
    
    -- Salva o nome de quem já estava selecionado antes de atualizar
    local nomeAtual = nil
    if selecionarTodos then
        nomeAtual = "Todos"
    elseif jogadorSelecionado then
        nomeAtual = jogadorSelecionado.Name
    end

    pcall(function()
        if DropdownJogadores and DropdownJogadores.Refresh then
            DropdownJogadores:Refresh(novasOpcoes)
        end
    end)
    
    -- Se o jogador ainda estiver no servidor, reaplica a seleção para não sumir
    if nomeAtual and table.find(novasOpcoes, nomeAtual) then
        pcall(function()
            if DropdownJogadores and DropdownJogadores.Set then
                DropdownJogadores:Set({nomeAtual})
            end
        end)
    else
        -- Se o jogador realmente saiu do jogo
        if jogadorSelecionado and not Players:FindFirstChild(jogadorSelecionado.Name) then
            SendNotification("⚠️ Alvo Perdido", jogadorSelecionado.Name .. " saiu do jogo!", 3)
            jogadorSelecionado = nil
            if espionando and LocalPlayer.Character then
                local hum = LocalPlayer.Character:FindFirstChildWhichIsA("Humanoid")
                if hum then Camera.CameraSubject = hum end
            end
            pcall(function()
                if DropdownJogadores and DropdownJogadores.Set then
                    DropdownJogadores:Set({"..."})
                end
            end)
        end
    end
end

PropsTab:CreateButton({

local function CriarSecaoModo(nomeModo, chaveConfig, altMin, altMax, altPadrao)

    PropsTab:CreateToggle({
        Name = "Ativar " .. nomeModo,
        CurrentValue = false,
        Flag = "ToggleModo" .. chaveConfig,
        Callback = function(s)
            PropMovementMode = s and chaveConfig or "Nenhum"
            if s then AtualizarCacheProps() end
        end
    })
    
    PropsTab:CreateSlider({
        Name = "Velocidade",
        Range = {1, 100},
        Increment = 1,
        CurrentValue = math.floor(Configs[chaveConfig].Speed * 20),
        Callback = function(v) Configs[chaveConfig].Speed = v / 20 end
    })
    
    PropsTab:CreateSlider({
        Name = "Distância",
        Range = {1, 50},
        Increment = 1,
        CurrentValue = Configs[chaveConfig].Distance,
        Callback = function(v) Configs[chaveConfig].Distance = v end
    })
    
    PropsTab:CreateSlider({
        Name = "Altura",
        Range = {altMin or -10, altMax or 30},
        Increment = 1,
        CurrentValue = altPadrao or Configs[chaveConfig].Height,
        Callback = function(v) Configs[chaveConfig].Height = v end
    })
end

-- LOOPS DE FUNCIONAMENTO (BACKGROUND)

-- 1. Gerenciamento automático da ferramenta PropMaker
task.spawn(function()
    local ferramentaNoInventario = false
    while true do
        if AlgumModoAtivo() then
            local char = LocalPlayer.Character
            local mochila = LocalPlayer:FindFirstChild("Backpack")
            local fE = char and char:FindFirstChild("PropMaker")
            local fM = mochila and mochila:FindFirstChild("PropMaker")
            
            if not fE and not fM then
                pcall(function()
                    ReplicatedStorage:WaitForChild("RE"):WaitForChild("1Too1l"):InvokeServer("PickingTools", "PropMaker")
                end)
            end
            ferramentaNoInventario = true
        else
            if ferramentaNoInventario then
                pcall(function()
                    local char = LocalPlayer.Character
                    local mochila = LocalPlayer:FindFirstChild("Backpack")
                    local fE = char and char:FindFirstChild("PropMaker")
                    local fM = mochila and mochila:FindFirstChild("PropMaker")
                    if fE then fE:Destroy() end
                    if fM then fM:Destroy() end
                end)
                ferramentaNoInventario = false
            end
        end
        task.wait(0.2)
    end
end)

-- 2. Atualização periódica do cache de Props
task.spawn(function()
    while true do

            pcall(AtualizarCacheProps)
        end
        task.wait(0.5)
    end
end)

-- 3. Loop Principal de Física e Movimentação
task.spawn(function()
    local angle, tempoGeral = 0, 0
    RunService.Heartbeat:Connect(function(dt)
        if (PropMovementMode == "Nenhum" and not ModoAvancadoAtivo) or #meusPropsCache == 0 then return end
        
        local lista = {}
        if selecionarTodos then
            for _, p in ipairs(Players:GetPlayers()) do
                if p ~= LocalPlayer and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
                    table.insert(lista, p)
                end
            end
        else
            local t = jogadorSelecionado or LocalPlayer
            if t.Character and t.Character:FindFirstChild("HumanoidRootPart") then
                table.insert(lista, t)
            end
        end
        
        if #lista == 0 then return end
        
        local cfg = Configs[PropMovementMode]
        local speedVal = cfg and cfg.Speed or 2
        angle = angle + (dt * speedVal)
        tempoGeral = tempoGeral + dt * (speedVal * 2)
        
        local total = #meusPropsCache
        local shift = selecionarTodos and math.floor(tick() * 30) or 0

        for i, prop in ipairs(meusPropsCache) do
            local remote = prop:FindFirstChild("SetCurrentCFrame")
            if remote then
                local pDono = lista[((i - 1 + shift) % #lista) + 1]
                if pDono and pDono.Character and pDono.Character:FindFirstChild("HumanoidRootPart") then
                    local root = pDono.Character.HumanoidRootPart
                    local rPos, rCF = root.Position, root.CFrame
                    local pFrente = rPos + (rCF.LookVector * 5)
                    local spacing = (math.pi * 2) / math.max(total, 1)
                    local fAngle = angle + (i * spacing) + (shift * 0.5)
                    local tPos = Vector3.zero
                    local olharPara = rPos

                    if ModoAvancadoAtivo then
                        local humAlvo = pDono.Character:FindFirstChildWhichIsA("Humanoid")
                        if humAlvo and (humAlvo.SeatPart or humAlvo.Sit) then
                            local posVoid = Vector3.new(rPos.X, -40000, rPos.Z)
                            task.spawn(function() pcall(remote.InvokeServer, remote, rCF - rCF.Position + posVoid) end)
                        else
                            local oscilacaoRapida = Vector3.new(0, math.sin(tempoGeral * 15 + i) * 3, 0)
                            task.spawn(function() pcall(remote.InvokeServer, remote, rCF + oscilacaoRapida) end)
                        end
                    elseif cfg then
                        if PropMovementMode == "Orbita" then
                            tPos = rPos + Vector3.new(math.cos(fAngle) * cfg.Distance, cfg.Height, math.sin(fAngle) * cfg.Distance)
                        elseif PropMovementMode == "Aura" then
                            local o = math.sin(tempoGeral * 3 + i) * 1.5
                            tPos = rPos + Vector3.new(math.cos(fAngle * 1.5) * (cfg.Distance + (o * 0.2)), cfg.Height + o, math.sin(fAngle * 1.5) * (cfg.Distance + (o * 0.2)))
                            olharPara = rPos + Vector3.new(0, o, 0)
                        elseif PropMovementMode == "Tremor" then
                            tPos = rPos + Vector3.new(math.cos(fAngle) * cfg.Distance, cfg.Height, math.sin(fAngle) * cfg.Distance) + Vector3.new(math.sin(tick() * 50 + i) * 0.8, math.cos(tick() * 55 + i) * 0.8, math.sin(tick() * 60 + i) * 0.8)
                        elseif PropMovementMode == "Caotico" then
                            tPos = rPos + Vector3.new(math.cos(fAngle) * cfg.Distance, cfg.Height, math.sin(fAngle) * cfg.Distance) + Vector3.new(math.sin(tempoGeral + (i * 2)) * (cfg.Distance * 0.5), math.cos(tempoGeral * 1.8 + i) * 4, math.sin(tempoGeral * 1.3 + i) * (cfg.Distance * 0.5))
                            olharPara = tPos + Vector3.new(math.sin(tempoGeral * 5), math.cos(tempoGeral * 5), math.sin(tempoGeral * 3))
                        elseif PropMovementMode == "Estrela" then
                            local e = cfg.Distance + (math.sin(tempoGeral * 2.5 + i) * (cfg.Distance * 0.4))
                            tPos = rPos + Vector3.new(math.cos(fAngle) * e, cfg.Height + (math.cos(fAngle) * 2), math.sin(fAngle) * e)
                            olharPara = rPos + Vector3.new(0, cfg.Height, 0)
                        elseif PropMovementMode == "Alinhamento" then
                            tPos = rPos + (rCF.LookVector * (-5 - (i * cfg.Distance))) + Vector3.new(0, cfg.Height + (math.sin(tempoGeral + i) * 0.5), 0)
                            olharPara = tPos - (rCF.LookVector * 100)
                        elseif PropMovementMode == "Furacao" then
                            local s = ((tempoGeral + (i * 2)) % 10) - 3
                            tPos = pFrente + Vector3.new(math.cos(fAngle * 1.5) * (cfg.Distance * (1 - (s / 15))), cfg.Height + s, math.sin(fAngle * 1.5) * (cfg.Distance * (1 - (s / 15))))
                            olharPara = pFrente + Vector3.new(0, 10, 0)
                        elseif PropMovementMode == "Portal" then
                            local rp = cfg.Distance * 0.7
                            tPos = (rCF * CFrame.new(0, cfg.Height + (math.sin(fAngle) * rp), -5)).Position + (rCF.RightVector * (math.cos(fAngle) * rp))
                            olharPara = pFrente + (rCF.LookVector * 20)
                        end
                        
                        task.spawn(function() pcall(remote.InvokeServer, remote, CFrame.lookAt(tPos, olharPara)) end)
                    end
                end
            end
        end
    end)
end)

-- 4. Loop RGB
task.spawn(function()
    while true do
        if PropsActive and #meusPropsCache > 0 then
            local tempoAtual = tick()
            local propColorRemote = LocalPlayer:FindFirstChild("PlayerGui")
                and LocalPlayer.PlayerGui:FindFirstChild("NoResetGUIHandler")
                and LocalPlayer.PlayerGui.NoResetGUIHandler:FindFirstChild("PropColorPicker")
                and LocalPlayer.PlayerGui.NoResetGUIHandler.PropColorPicker:FindFirstChild("SetColor")

            for i, prop in ipairs(meusPropsCache) do
                local remote = prop:FindFirstChild("ChangePropColor")
                if remote then
                    local bp = prop:FindFirstChildWhichIsA("BasePart")
                    local color = Color3.fromHSV(((tempoAtual * WaveSpeed) + ((bp and bp.Position.X or i) * ColorSpread)) % 1, 1, 1)
                    for _, part in ipairs(prop:GetDescendants()) do
                        if part:IsA("BasePart") then part.Color = color end
                    end
                    task.spawn(function() pcall(remote.InvokeServer, remote, color) end)
                end
            end
            
            if propColorRemote then
                pcall(function()
                    propColorRemote:FireServer(Color3.fromHSV((tempoAtual * WaveSpeed) % 1, 1, 1))
                end)
            end
        end
        task.wait(0.05)
    end
end)
