-- --- FUNÇÃO ANTI-LAG ---
local backupTables = { Vehicles = {}, Canoes = {}, Jets = {}, Helis = {} }
local function AntiFlingLoop(name, getFolderFunc)
    local active = false
    task.spawn(function()
        while true do
            if active and vu7.Character then
                local folder = getFolderFunc()
                if folder then
                    for _, item in ipairs(folder:GetChildren()) do
                        local isMine = false
                        if name == "Vehicles" then
                            for _, seat in ipairs(item:GetDescendants()) do
                                if (seat:IsA("VehicleSeat") or seat:IsA("Seat")) and seat.Occupant and seat.Occupant.Parent == vu7.Character then
                                    isMine = true
                                    break
                                end
                            end
                        elseif name == "Canoes" then
                            local owner = item:FindFirstChild("Owner")
                            isMine = owner and owner.Value == vu7
                        elseif name == "Jets" or name == "Helis" then
                            isMine = item.Name == vu7.Name
                        end
                        if not isMine then
                            table.insert(backupTables[name], item:Clone())
                            item:Destroy()
                        end
                    end
                end
            end
            task.wait(0.03)
        end
    end)
    return function(state)
        active = state
        if not state then
            for _, item in ipairs(backupTables[name]) do
                local parentFolder = getFolderFunc()
                if parentFolder then item.Parent = parentFolder end
            end
            backupTables[name] = {}
        end
    end
end

-- --- FUNÇÕES HEADSIT ---
local function headsitOnPlayer(targetPlayer)
    if not targetPlayer or not targetPlayer.Character then return end
    local char = vu7.Character
    local root = char:FindFirstChild("HumanoidRootPart")
    local tHead = targetPlayer.Character:FindFirstChild("Head")
    if root and tHead then
        root.CFrame = tHead.CFrame * CFrame.new(0, 2.2, 0)
        local weld = Instance.new("WeldConstraint", root)
        weld.Part0 = root
        weld.Part1 = tHead
        char.Humanoid.Sit = true
    end
end

local function removeHeadsit()
    if vu7.Character and vu7.Character:FindFirstChild("HumanoidRootPart") then
        for _, v in pairs(vu7.Character.HumanoidRootPart:GetChildren()) do
            if v:IsA("WeldConstraint") then v:Destroy() end
        end
        vu7.Character.Humanoid.Sit = false
    end
end

-- Forçar partes (Física do Black Hole)
local function ForcePart(v)
    if v:IsA("BasePart") and not v.Anchored and not v.Parent:FindFirstChildOfClass("Humanoid") and v.Name ~= "Handle" then
        for _, x in ipairs(v:GetChildren()) do
            if x:IsA("BodyMover") or x:IsA("AlignPosition") or x:IsA("Torque") then x:Destroy() end
        end
        v.CanCollide = false
        local torque = Instance.new("Torque", v)
        torque.Torque = Vector3.new(100000000, 1000000000, 100000000)
        local att0 = Instance.new("Attachment", v)
        torque.Attachment0 = att0
        local alignPos = Instance.new("AlignPosition", v)
        alignPos.Name = "BH_Force"
        alignPos.MaxForce = math.huge
        alignPos.MaxVelocity = math.huge
        alignPos.Responsiveness = _G.BH_Strength
        alignPos.Attachment0 = att0
        alignPos.Attachment1 = BH_Attachment
    end
end

local function FlingBall(target)
	local player = eu
	local character = player.Character or player.CharacterAdded:Wait()
	local humanoid = character:WaitForChild("Humanoid")
	local hrp = character:WaitForChild("HumanoidRootPart")
	local backpack = player:WaitForChild("Backpack")
	local ServerBalls = mundo.WorkspaceCom:FindFirstChild("001_SoccerBalls")
	if not ServerBalls then return end

	local function GetBall()
		if not backpack:FindFirstChild("SoccerBall") and not character:FindFirstChild("SoccerBall") then
			rep.RE:FindFirstChild("1Too1l"):InvokeServer("PickingTools", "SoccerBall")
		end

		repeat task.wait() until backpack:FindFirstChild("SoccerBall") or character:FindFirstChild("SoccerBall")

		local ballTool = backpack:FindFirstChild("SoccerBall")
		if ballTool then ballTool.Parent = character end

		repeat task.wait() until ServerBalls:FindFirstChild("Soccer" .. player.Name)
		return ServerBalls:FindFirstChild("Soccer" .. player.Name)
	end

	local Ball = ServerBalls:FindFirstChild("Soccer" .. player.Name) or GetBall()
	if not Ball then return end

	Ball.CanCollide = false
	Ball.Massless = true
	Ball.CustomPhysicalProperties = PhysicalProperties.new(0.0001, 0, 0)

	if target ~= player then
		local tchar = target.Character
		if not tchar then return end

		local troot = tchar:FindFirstChild("HumanoidRootPart")
		local thum = tchar:FindFirstChild("Humanoid")
		if not troot or not thum then return end

		local bvAntigo = Ball:FindFirstChildWhichIsA("BodyVelocity")
		if bvAntigo then bvAntigo:Destroy() end

		local bv = Instance.new("BodyVelocity")
		bv.Name = "FlingPower"
		bv.Velocity = Vector3.new(1e9, 1e9, 1e9)
		bv.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
		bv.P = 9e900
		bv.Parent = Ball

		repeat
			local pos = troot.Position + (troot.Velocity / 1.5)
			local tempo = tick()

		
			local altura = math.sin(tempo * 7) * 2
			Ball.CFrame = CFrame.new(pos.X, pos.Y + altura, pos.Z)
			Ball.Orientation += Vector3.new(45, 60, 30)

			task.wait(1 / 5000)
		until troot.Velocity.Magnitude > 1000 or thum.Health <= 0 or not tchar:IsDescendantOf(mundo) or target.Parent ~= jogadores
	end
end

-- Botão para remover ban de todas as casas
CasasTab:AddButton({
    Name = "Remover Ban de Todas as Casas Manual",
    Description = "Tenta remover o ban de todas as casas ",
    Callback = function()
        local successCount = 0
        local failCount = 0
        for i = 1, 37 do
            local bannedBlockName = "BannedBlock" .. i
            local bannedBlock = Workspace:FindFirstChild(bannedBlockName, true)
            if bannedBlock then
                local success, _ = pcall(function()
                    bannedBlock:Destroy()
                end)
                if success then
                    successCount = successCount + 1
                else
                    failCount = failCount + 1
                end
            end
        end
        for _, house in pairs(Workspace:GetDescendants()) do
            if house.Name:match("BannedBlock") then
                local success, _ = pcall(function()
                    house:Destroy()
                end)
                if success then
                    successCount = successCount + 1
                else
                    failCount = failCount + 1
                end
            end
        end
        if successCount > 0 then
            game.StarterGui:SetCore("SendNotification", {
                Title = "Sucesso",
                Text = "Bans removidos de " .. successCount .. " casas!",
                Duration = 5
            })
        end
        if failCount > 0 then
            game.StarterGui:SetCore("SendNotification", {
                Title = "Aviso",
                Text = "Falha ao remover bans de " .. failCount .. " casas.",
                Duration = 5
            })
        end
        if successCount == 0 and failCount == 0 then
            game.StarterGui:SetCore("SendNotification", {
                Title = "Aviso",
                Text = "Nenhum ban encontrado para remover.",
                Duration = 5
            })
        end
    end
})
