	local Box = Instance.new("TextBox")
	Box.Parent = Frame
	Box.Size = UDim2.new(0.28, 0, 0.65, 0)
	Box.Position = UDim2.new(0.68, 0, 0.18, 0)
	Box.BackgroundColor3 = Color3.fromRGB(35, 35, 50)
	Box.Text = tostring(default)
	Box.TextColor3 = Color3.fromRGB(0, 255, 180)
	Box.Font = Enum.Font.SourceSansBold
	Box.TextSize = 12

	local BoxCorner = Instance.new("UICorner")
	BoxCorner.CornerRadius = UDim.new(0, 4)
	BoxCorner.Parent = Box

	Box.FocusLost:Connect(function()
		local val = tonumber(Box.Text)
		if val then
			val = math.clamp(val, min, max)
			Box.Text = tostring(val)
			callback(val)
		else
			Box.Text = tostring(default)
		end
	end)
end

local function GetAimbotTarget()
	local selectedTarget = nil
	local shortestDist = math.huge
	local mousePos = UserInputService:GetMouseLocation()

	for _, plr in pairs(Players:GetPlayers()) do
		if plr ~= LocalPlayer and plr.Character then
			local isMatchingName = false
			if AimbotTargetUsername ~= "" then
				if plr.Name:lower():find(AimbotTargetUsername, 1, true) or plr.DisplayName:lower():find(AimbotTargetUsername, 1, true) then
					isMatchingName = true
				end
			else
				isMatchingName = true
			end

			if isMatchingName then
				local targetPart = plr.Character:FindFirstChild(AimbotPart) or plr.Character:FindFirstChild("HumanoidRootPart")
				local hum = plr.Character:FindFirstChildOfClass("Humanoid")

				if targetPart and hum and hum.Health > 0 then
					local screenPos, onScreen = Camera:WorldToViewportPoint(targetPart.Position)
					if onScreen then
						local dist = (Vector2.new(screenPos.X, screenPos.Y) - mousePos).Magnitude
						if dist <= AimbotFOVRadius and dist < shortestDist then
							if AimbotWallCheck then
								local ray = Camera:ViewportPointToRay(screenPos.X, screenPos.Y)
								local raycastResult = Workspace:Raycast(ray.Origin, ray.Direction * 1000)
								if raycastResult and raycastResult.Instance:IsDescendantOf(plr.Character) then
									selectedTarget = targetPart
									shortestDist = dist
								end
							else
								selectedTarget = targetPart
								shortestDist = dist
							end
						end
					end
				end
			end
		end
	end
	return selectedTarget
end

		end

		local targetPart = GetAimbotTarget()
		if targetPart then
			local targetPosition = targetPart.Position
			if AimbotPredictMovement and targetPart.Parent:FindFirstChild("HumanoidRootPart") then
				targetPosition = targetPosition + (targetPart.Parent.HumanoidRootPart.AssemblyLinearVelocity * AimbotPredictionFactor)
			end

			local currentCameraCFrame = Camera.CFrame
			local targetCameraCFrame = CFrame.new(currentCameraCFrame.Position, targetPosition)
			Camera.CFrame = currentCameraCFrame:Lerp(targetCameraCFrame, math.clamp(1 / AimbotSmoothness, 0.01, 1))

			if TriggerBotEnabled then
				local tool = LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Tool")
				if tool then
					tool:Activate()
				end
			end
		end
	end

	-- ESP Visual System Updates
	for _, plr in pairs(Players:GetPlayers()) do
		if plr ~= LocalPlayer and plr.Character then
			local highlight = plr.Character:FindFirstChild("HubESP")
			if ESPEnabled then
				if not highlight then
					highlight = Instance.new("Highlight")
					highlight.Name = "HubESP"
					highlight.FillColor = Color3.fromRGB(0, 255, 180)
					highlight.OutlineColor = Color3.fromRGB(255, 255, 255)
					highlight.Parent = plr.Character
				end
			else
				if highlight then highlight:Destroy() end
			end
		end
	end
end)

-- 14. CORE ENGINE HEARTBEAT (PHYSICS, FLING, & SAFEGUARD LOOP)
-----------------------------------------------------------------------
RunService.Heartbeat:Connect(function(deltaTime)
	local char = LocalPlayer.Character
	if not char then return end

	local hum = char:FindFirstChildOfClass("Humanoid")
	local hrp = char:FindFirstChild("HumanoidRootPart")

	-------------------------------------------------------------------
	-- 🛡️ ANTI-SELF-FLING & VELOCITY LOCK SAFEGUARD
	-------------------------------------------------------------------
	if hrp then
		if AntiFlingEnabled then
			if hrp.AssemblyAngularVelocity.Magnitude > 80 and not VoidFlingEnabled and not SpinBotEnabled then
				hrp.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
			end
			if hrp.AssemblyLinearVelocity.Magnitude > 300 and not FlyEnabled then
				hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
			end
		end

		if FreezeInAir then
			hrp.Anchored = true
		elseif not AnchorSelf then
			hrp.Anchored = false
		end

		if AnchorSelf then
			hrp.Anchored = true
		end
	end

	-------------------------------------------------------------------
	-- HUMANOID STATE CONTROLS
	-------------------------------------------------------------------
	if hum then
		hum.WalkSpeed = WalkSpeedValue
		hum.UseJumpPower = true
		hum.JumpPower = JumpPowerValue
		hum.HipHeight = HipHeightValue

		if LowGravity then
			Workspace.Gravity = 30
		else
			Workspace.Gravity = 196.2
		end

		if AutoBhop and hum.MoveDirection.Magnitude > 0 then
			if hum.FloorMaterial ~= Enum.Material.Air then
				hum:ChangeState(Enum.HumanoidStateType.Jumping)
			end
		end

		if AntiSeatingSelf then
			hum.Sit = false
		end

		if AntiRagdollAbsolute then
			local state = hum:GetState()
			if state == Enum.HumanoidStateType.Ragdoll or state == Enum.HumanoidStateType.Physics or state == Enum.HumanoidStateType.FallingDown then
				hum:ChangeState(Enum.HumanoidStateType.GettingUp)
			end
		end

		if SwimInAir then
			hum:ChangeState(Enum.HumanoidStateType.Swimming)
		end

		if AutoRespawn and hum.Health <= 0 then
			LocalPlayer:Unload()
		end
	end

	-------------------------------------------------------------------
	-- FTAP ANTI-GRAB & ANTI-ATTACH
	-------------------------------------------------------------------
	if AntiGrabEnabled or AntiAttachEnabled then
		for _, item in pairs(char:GetDescendants()) do
			if item:IsA("WeldConstraint") or item:IsA("Weld") or item:IsA("RopeConstraint") or item:IsA("AlignPosition") or item:IsA("SpringConstraint") then
				if item:IsA("WeldConstraint") or item:IsA("Weld") then
					if (item.Part0 and item.Part0.Parent ~= char) or (item.Part1 and item.Part1.Parent ~= char) then
						item:Destroy()
					end
				else
					item:Destroy()
				end
			end
		end
	end

	-------------------------------------------------------------------
	-- GOD MODE LOGIC
	-------------------------------------------------------------------
	if GodModeFE and char:FindFirstChild("LowerTorso") and char.LowerTorso:FindFirstChild("Root") then
		char.LowerTorso.Root:Destroy()
	end

	-------------------------------------------------------------------
	-- ANTI-VOID PROTECTION
	-------------------------------------------------------------------
	if AntiVoidEnabled and hrp then
		if hrp.Position.Y < -50 then
			hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
			hrp.CFrame = CFrame.new(hrp.Position.X, 50, hrp.Position.Z)
		end
	end

	-------------------------------------------------------------------
	-- VOID FLING & SPIN BOT
	-------------------------------------------------------------------
	if VoidFlingEnabled and hrp then
		hrp.AssemblyAngularVelocity = Vector3.new(0, 999999999, 0)
	elseif SpinBotEnabled and hrp then
		hrp.AssemblyAngularVelocity = Vector3.new(0, SpinSpeed * 10, 0)
	end

	-------------------------------------------------------------------
	-- KILL AURA & PHYSICAL ATTACKS
	-------------------------------------------------------------------
	if (KillAuraEnabled or MassSitAura or RapidPunchAura or SuperRagdollAll or PushAuraFE or AirJugglerAura or FreezeAura or BlackHoleVoid) and hrp and hum then
		for _, plr in pairs(Players:GetPlayers()) do
			if plr ~= LocalPlayer and plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") then
				local enemyHum = plr.Character:FindFirstChildOfClass("Humanoid")
				local enemyHrp = plr.Character.HumanoidRootPart

				if enemyHum and enemyHum.Health > 0 then
					local distance = (hrp.Position - enemyHrp.Position).Magnitude
					if distance <= KillAuraRadius then
						if AutoActivateTool then
							local tool = char:FindFirstChildOfClass("Tool")
							if tool then tool:Activate() end
						end

						if PushAuraFE then
							enemyHrp.AssemblyLinearVelocity = (enemyHrp.Position - hrp.Position).Unit * 2500
						elseif AirJugglerAura then
							enemyHrp.AssemblyLinearVelocity = Vector3.new(0, 2000, 0)
						elseif FreezeAura then
							enemyHrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
						elseif MassSitAura or ForcedSitEnabled or SuperRagdollAll then
							enemyHrp.CFrame = hrp.CFrame * CFrame.new(0, -1.8, -0.5)
							enemyHrp.AssemblyLinearVelocity = Vector3.new(0, -1000, 0)
						elseif AuraTripForce then
							enemyHrp.CFrame = hrp.CFrame * CFrame.Angles(math.rad(90), 0, 0)
							enemyHrp.AssemblyLinearVelocity = Vector3.new(0, -500, 0)
						elseif TeleportBehind then
							hrp.CFrame = enemyHrp.CFrame * CFrame.new(0, 0, 2)
						end

						if BlackHoleVoid then
							enemyHrp.AssemblyLinearVelocity = Vector3.new(0, -50000, 0)
						end
					end
				end
			end
		end
	end

	-------------------------------------------------------------------
	-- TARGET LOOP FLING ENGINE
	-------------------------------------------------------------------
	if LoopFlingEnabled and LoopFlingTarget ~= "" and hrp then
		for _, plr in pairs(Players:GetPlayers()) do
			if plr ~= LocalPlayer and (plr.Name:lower():find(LoopFlingTarget, 1, true) or plr.DisplayName:lower():find(LoopFlingTarget, 1, true)) then
				if plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") then
					local targetHrp = plr.Character.HumanoidRootPart
					hrp.CFrame = targetHrp.CFrame * CFrame.new(0, 0, 1.2)
					hrp.AssemblyAngularVelocity = Vector3.new(999999, 999999, 999999)
					targetHrp.AssemblyLinearVelocity = Vector3.new(0, -20000, 0)
				end
			end
		end
	end

	-------------------------------------------------------------------
	-- NOCLIP EXECUTION
	-------------------------------------------------------------------
	if NoclipEnabled then
		for _, part in pairs(char:GetDescendants()) do
			if part:IsA("BasePart") then
				if not IsOtherPlayerPart(part) then
					part.CanCollide = false
				end
			end
		end
	end

	-------------------------------------------------------------------
	-- VISUAL & CHAOS EFFECTS
	-------------------------------------------------------------------
	if RainbowPlayer then
		for _, part in pairs(char:GetChildren()) do
			if part:IsA("BasePart") then
				part.Color = Color3.fromHSV(tick() % 2 / 2, 1, 1)
			end
		end
	end

	if SpasmCharacter and hrp then
		hrp.CFrame = hrp.CFrame * CFrame.Angles(math.rad(math.random(-30, 30)), math.rad(math.random(-30, 30)), math.rad(math.random(-30, 30)))
	end

	if HeadSpin and char:FindFirstChild("Head") then
		char.Head.CFrame = char.Head.CFrame * CFrame.Angles(0, math.rad(20), 0)
	end

	-------------------------------------------------------------------
	-- LIGHTING CONTROL ENGINE
	-------------------------------------------------------------------
	if FullbrightEnabled or NightVision then
		Lighting.Brightness = 3
		Lighting.ClockTime = ChangeTime
		Lighting.GlobalShadows = false
	end
	if NoFogEnabled then
		Lighting.FogEnd = 100000
	end
end)

-----------------------------------------------------------------------
-- 15. FLY ENGINE CONTROLLER
-----------------------------------------------------------------------
RunService.RenderStepped:Connect(function()
	local char = LocalPlayer.Character
	if char and char:FindFirstChild("HumanoidRootPart") then
		local hrp = char.HumanoidRootPart

		if FlyEnabled then
			if not hrp:FindFirstChild("FlyVel") then
				local bodyVel = Instance.new("BodyVelocity")
				bodyVel.Name = "FlyVel"
				bodyVel.MaxForce = Vector3.new(1, 1, 1) * 10^7
				bodyVel.Parent = hrp

				local bodyGyro = Instance.new("BodyGyro")
				bodyGyro.Name = "FlyGyro"
				bodyGyro.MaxTorque = Vector3.new(1, 1, 1) * 10^7
				bodyGyro.P = 10000
				bodyGyro.Parent = hrp
			end

			hrp.FlyGyro.CFrame = Camera.CFrame
			local moveDir = Vector3.new()
			if UserInputService:IsKeyDown(Enum.KeyCode.W) then moveDir = moveDir + Camera.CFrame.LookVector end
			if UserInputService:IsKeyDown(Enum.KeyCode.S) then moveDir = moveDir - Camera.CFrame.LookVector end
			if UserInputService:IsKeyDown(Enum.KeyCode.A) then moveDir = moveDir - Camera.CFrame.RightVector end
			if UserInputService:IsKeyDown(Enum.KeyCode.D) then moveDir = moveDir + Camera.CFrame.RightVector end
			if UserInputService:IsKeyDown(Enum.KeyCode.Space) then moveDir = moveDir + Vector3.new(0, 1, 0) end
			if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then moveDir = moveDir - Vector3.new(0, 1, 0) end

			hrp.FlyVel.Velocity = moveDir * FlySpeedValue
		else
			if hrp:FindFirstChild("FlyVel") then hrp.FlyVel:Destroy() end
			if hrp:FindFirstChild("FlyGyro") then hrp.FlyGyro:Destroy() end
		end
	end
end)
