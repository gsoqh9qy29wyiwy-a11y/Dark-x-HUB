local function createLoadingScreen()
    local promise = {}
    promise.completed = false
    promise.connection = nil
    
    pcall(function()
        if CoreGui:FindFirstChild("LoadingScreen") then
            CoreGui.LoadingScreen:Destroy()
        end
    end)

    local gui = Instance.new("ScreenGui")
    gui.Name = "LoadingScreen"
    gui.IgnoreGuiInset = true
    gui.ResetOnSpawn = false
    gui.Parent = CoreGui
    pcall(function() syn.protect_gui(gui) end)

    local frame = Instance.new("Frame")
    frame.BackgroundColor3 = Color3.new(0, 0, 0)
    frame.BackgroundTransparency = 0.0
    frame.Size = UDim2.new(1, 0, 1, 0)
    frame.Parent = gui

    local image = Instance.new("ImageLabel")
    image.Size = UDim2.new(0, 300, 0, 300)
    image.Position = UDim2.new(0.5, -150, 0.35, -150)
    image.BackgroundTransparency = 1
    image.Image = "rbxassetid://131153193945220"
    image.Parent = frame

    local nameText = Instance.new("TextLabel")
    nameText.Size = UDim2.new(1, 0, 0, 60)
    nameText.Position = UDim2.new(0, 0, 0.65, 0)
    nameText.TextColor3 = Color3.fromRGB(255, 0, 0)
    nameText.TextStrokeTransparency = 0.3
    nameText.BackgroundTransparency = 1
    nameText.Font = Enum.Font.FredokaOne
    nameText.TextScaled = true
    nameText.Text = "Verificando usuário: "..lp.Name.." (Conta: "..tostring(lp.AccountAge).." dias)"
    nameText.Parent = frame

    local barBG = Instance.new("Frame")
    barBG.Size = UDim2.new(0.6, 0, 0.03, 0)
    barBG.Position = UDim2.new(0.2, 0, 0.8, 0)
    barBG.BackgroundColor3 = Color3.new(0, 0, 0)
    barBG.BorderSizePixel = 9
    barBG.Parent = frame

    local bar = Instance.new("Frame")
    bar.Size = UDim2.new(0, 0, 1, 0)
    bar.BackgroundColor3 = Color3.new(255, 0, 0)
    bar.BorderSizePixel = 9
    bar.Parent = barBG

    local tween = TweenService:Create(
        bar,
        TweenInfo.new(4, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
        {Size = UDim2.new(1, 0, 1, 0)}
    )
    tween:Play()

    local function cleanup()
        if gui then gui:Destroy() end
        if promise.connection then promise.connection:Disconnect() end
    end

    promise.connection = tween.Completed:Connect(function()
        TweenService:Create(frame, TweenInfo.new(1, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {BackgroundTransparency = 1}):Play()
        TweenService:Create(barBG, TweenInfo.new(1, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {BackgroundTransparency = 1}):Play()
        TweenService:Create(bar, TweenInfo.new(1, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {BackgroundTransparency = 1}):Play()
        TweenService:Create(image, TweenInfo.new(1, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {ImageTransparency = 1}):Play()
        TweenService:Create(nameText, TweenInfo.new(1, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {TextTransparency = 1}):Play()
        
        task.delay(1.2, function()
            local sound = Instance.new("Sound")
            sound.SoundId = "rbxassetid://113088956809836"
            sound.Volume = 0.5
            sound.PlayOnRemove = true
            sound.Parent = workspace
            sound:Destroy()
            
            StarterGui:SetCore("SendNotification", {
                Title = "Dark x Hub",
                Text = "Bem-vindo à Dark x Hub",
                Duration = 4
            })
            
            promise.completed = true
            cleanup()
        end)
    end)
    
    return promise
end

local function waitForPromise(promise)
    while not promise.completed do
        task.wait()
    end
end

local function EquiparSofaGarantido()
    local character = vu7.Character or vu7.CharacterAdded:Wait()
    local humanoid = character:WaitForChild("Humanoid", 3)
    local backpack = vu7:WaitForChild("Backpack", 3)
    if not humanoid or not backpack then return false end

    if character:FindFirstChild("Couch") then return true end

    if not backpack:FindFirstChild("Couch") then
        pcall(function()
            local toolRemote = vu8:FindFirstChild("RE") and (vu8.RE:FindFirstChild("1Too1l") or vu8.RE:FindFirstChild("1Tool"))
            if toolRemote then
                toolRemote:InvokeServer("PickingTools", "Couch")
            end
        end)
        
        local t0 = tick()
        repeat
            task.wait(0.05)
        until backpack:FindFirstChild("Couch") or (tick() - t0 > 1.5)
    end

    local couchTool = backpack:FindFirstChild("Couch")
    if couchTool then
        humanoid:EquipTool(couchTool)
        task.wait(0.1)
        return true
    end

    return character:FindFirstChild("Couch") ~= nil
end

local function ExecutarBringCouch(targetPlayer)
    if not targetPlayer or not targetPlayer.Character or not vu7.Character then return end

    local character = vu7.Character
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    local myHRP = character:FindFirstChild("HumanoidRootPart")
    local tHRP = targetPlayer.Character:FindFirstChild("HumanoidRootPart")
    local tHum = targetPlayer.Character:FindFirstChild("Humanoid")

    if not humanoid or not myHRP or not tHRP or not tHum then return end

    local originalCFrame = myHRP.CFrame
    local pegou = EquiparSofaGarantido()
    if not pegou then return end

    local running = true
    local startTime = tick()
    local bringLoop

    local noCollide = vu10.Stepped:Connect(function()
        if not running then return end
        for _, part in ipairs(character:GetDescendants()) do
            if part:IsA("BasePart") then part.CanCollide = false end
        end
    end)

    bringLoop = vu10.Heartbeat:Connect(function()
        if not running or not targetPlayer.Character or not tHRP or not myHRP then
            if bringLoop then bringLoop:Disconnect() end
            if noCollide then noCollide:Disconnect() end
            return
        end

        pcall(function()
            local targetVel = tHRP.AssemblyLinearVelocity or tHRP.Velocity
            local isMoving = (tHum.MoveDirection.Magnitude > 0.005) or (targetVel.Magnitude > 3.5)
            local finalPos = isMoving and (tHRP.CFrame * CFrame.new(0, 0, - ( (math.sin(tick() * 9000) + 40) / 800 * 500.5 ))).Position or tHRP.Position
            local rotation = CFrame.Angles(
                math.rad(vu9.DistributedGameTime * 1009400000),
                math.rad(vu9.DistributedGameTime * 19440000000),
                math.rad(vu9.DistributedGameTime * 4990900000)
            )
            myHRP.CFrame = CFrame.new(finalPos) * rotation
        end)

        if tHum.Sit or (tick() - startTime > 1) then
            running = false
            bringLoop:Disconnect()
            noCollide:Disconnect()

            pcall(function()
                myHRP.Anchored = true
                myHRP.Velocity = Vector3.zero
                myHRP.RotVelocity = Vector3.zero
                myHRP.CFrame = originalCFrame
            end)

            task.wait(0.25)
            pcall(function()
                humanoid:UnequipTools()
                local clearRemote = vu8:FindFirstChild("RE") and (vu8.RE:FindFirstChild("1Clea1rToo1ls") or vu8.RE:FindFirstChild("1Clea1rTool1s"))
                if clearRemote then
                    clearRemote:FireServer("PlayerWantsToDeleteTool", "Couch")
                end
            end)

            task.wait(0.1)
            pcall(function()
                myHRP.Velocity = Vector3.zero
                myHRP.RotVelocity = Vector3.zero
                myHRP.Anchored = false
                humanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
            end)
        end
    end)
    
    repeat task.wait(0.05) until running == false
end

-- --- FUNÇÃO ANTI-LAG ---
local backupTables = { Vehicles = {}, Canoes = {}, Jets = {}, Helis = {}, Balls = {} }
local function AntiFlingLoop(name, getFolderFunc)
    local active = false
    task.spawn(function()
        while true do
            if active and vu7.Character then
                local folder = getFolderFunc()
                if folder then
                    for _, item in ipairs(folder:GetChildren()) do
                        local isMine = false
                        if name == "Vehicles" then
                            for _, seat in ipairs(item:GetDescendants()) do
                                if (seat:IsA("VehicleSeat") or seat:IsA("Seat")) and seat.Occupant and seat.Occupant.Parent == vu7.Character then
                                    isMine = true
                                    break
                                end
                            end
                        elseif name == "Canoes" then
                            local owner = item:FindFirstChild("Owner")
                            isMine = owner and owner.Value == vu7
                        elseif name == "Jets" or name == "Helis" then
                            isMine = item.Name == vu7.Name
                        end
                        if not isMine then
                            table.insert(backupTables[name], item:Clone())
                            item:Destroy()
                        end
                    end
                end
            end
            task.wait(0.03)
        end
    end)
    return function(state)
        active = state
        if not state then
            for _, item in ipairs(backupTables[name]) do
                local parentFolder = getFolderFunc()
                if parentFolder then item.Parent = parentFolder end
            end
            backupTables[name] = {}
        end
    end
end

local function FlingBall(targetName)
    local target = game:GetService("Players"):FindFirstChild(targetName)
    if not target then return end
    local character = vu7.Character or vu7.CharacterAdded:Wait()
    local humanoid = character:WaitForChild("Humanoid")
    local backpack = vu7:WaitForChild("Backpack")
    local ServerBalls = vu9.WorkspaceCom:WaitForChild("001_SoccerBalls")
    
    local function GetBallInternal()
        if not backpack:FindFirstChild("SoccerBall") then
            vu8.RE:FindFirstChild("1Too1l"):InvokeServer("PickingTools", "SoccerBall")
        end
        repeat task.wait() until backpack:FindFirstChild("SoccerBall")
        backpack.SoccerBall.Parent = character
        repeat task.wait() until ServerBalls:FindFirstChild("Soccer" .. vu7.Name)
        character.SoccerBall.Parent = backpack
        return ServerBalls:FindFirstChild("Soccer" .. vu7.Name)
    end

    local Ball = ServerBalls:FindFirstChild("Soccer" .. vu7.Name) or GetBallInternal()
    Ball.CanCollide = false
    Ball.Massless = true
    Ball.CustomPhysicalProperties = PhysicalProperties.new(0.0001, 0, 0)

    if target ~= vu7 then
        local tchar = target.Character
        if tchar and tchar:FindFirstChild("HumanoidRootPart") and tchar:FindFirstChild("Humanoid") then
            local troot = tchar.HumanoidRootPart
            local thum = tchar.Humanoid
            
            if Ball:FindFirstChildWhichIsA("BodyVelocity") then
                Ball:FindFirstChildWhichIsA("BodyVelocity"):Destroy()
            end
            
            local bv = Instance.new("BodyVelocity")
            bv.Name = "FlingPower"
            bv.Velocity = Vector3.new(9e8, 9e8, 9e8)
            bv.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
            bv.P = 9e900
            bv.Parent = Ball
            
            vuCam.CameraSubject = thum
            
            local connection
            connection = vu10.Heartbeat:Connect(function()
                if not tchar or not tchar:IsDescendantOf(workspace) or thum.Health <= 0 or target.Parent ~= vu6 or troot.Velocity.Magnitude > 1000 then

local function sendChatCommand(command)
    if not PossuiPermissaoAdmin() then
        SendNotification("Acesso Negado", "Você não tem permissão para acessar o painel", 3)
        return
    end

    if not command then return end
    task.wait(0.1)
    local success = false
    if TextChatService and TextChatService.ChatVersion == Enum.ChatVersion.TextChatService then
        local channel = TextChatService:FindFirstChild("TextChannels")
        if channel then
            local generalChannel = channel:FindFirstChild("RBXGeneral")
            if generalChannel then
                local ok = pcall(function()
                    generalChannel:SendAsync(command)
                end)
                if ok then success = true end
            end
        end
    end
    if not success then
        local defaultChatSystemChatEvents = vu8:FindFirstChild("DefaultChatSystemChatEvents")
        if defaultChatSystemChatEvents then
            local sayMessageRequest = defaultChatSystemChatEvents:FindFirstChild("SayMessageRequest")
            if sayMessageRequest then
                pcall(function()
                    sayMessageRequest:FireServer(command, "All")
                    success = true
                end)
            end
        end
    end
    if not success and game:GetService("Chat") then
        pcall(function()
            game:GetService("Chat"):Chat(vu7.Character and vu7.Character:FindFirstChild("Head"), command)
            success = true
        end)
    end
    
    if success then
        SendNotification("Sucesso", "Comando enviado: " .. command, 3)
    else
        SendNotification("Erro", "Falha ao enviar: " .. command, 3)
    end
end

local function CreateJail(player)
    if Jajail[player] then return end
    local character = player.Character
    if character and character:FindFirstChild("HumanoidRootPart") then
        local HRP = character.HumanoidRootPart
        local cageSize = Vector3.new(10, 10, 10)
        local wallThickness = 0.5
        local transparency = 0.3
        local cageColor = Color3.fromRGB(75, 0, 130)

        local function buildCage(targetHRP, suffix)
            local cageModel = Instance.new("Model")
            cageModel.Name = "JaulaDeAdmin_" .. player.Name .. suffix
            cageModel.Parent = vu9

            local function createWall(size, positionOffset)
                local wall = Instance.new("Part")
                wall.Size = size
                wall.Anchored = true
                wall.CanCollide = true
                wall.Transparency = transparency
                wall.Color = cageColor
                wall.Material = Enum.Material.ForceField
                wall.CFrame = targetHRP.CFrame * positionOffset
                wall.Parent = cageModel

                local stroke = Instance.new("SelectionBox")
                stroke.Color3 = Color3.fromRGB(147, 112, 219)
                stroke.LineThickness = 0.2
                stroke.Transparency = 0.5
                stroke.Adornee = wall
                stroke.Parent = wall

                local light = Instance.new("PointLight")
                light.Color = Color3.fromRGB(147, 112, 219)
                light.Brightness = 0.5
                light.Range = 15
                light.Parent = wall
            end

            createWall(Vector3.new(cageSize.X, wallThickness, cageSize.Z), CFrame.new(0, -cageSize.Y/2, 0))
            createWall(Vector3.new(cageSize.X, wallThickness, cageSize.Z), CFrame.new(0, cageSize.Y/2, 0))
            createWall(Vector3.new(cageSize.X, cageSize.Y, wallThickness), CFrame.new(0, 0, -cageSize.Z/2))
            createWall(Vector3.new(cageSize.X, cageSize.Y, wallThickness), CFrame.new(0, 0, cageSize.Z/2))
            createWall(Vector3.new(wallThickness, cageSize.Y, cageSize.Z), CFrame.new(cageSize.X/2, 0, 0))
            createWall(Vector3.new(wallThickness, cageSize.Y, cageSize.Z), CFrame.new(-cageSize.X/2, 0, 0))

            return cageModel
        end

        local jailModel = buildCage(HRP, "_Preso")
        Jajail[player] = jailModel
        SendNotification("Foi preso", player.Name .. " foi preso na jaula", 4)
    end
end

local function FreezePlayer(player)
    if MagGelion[player] then return end
    local character = player.Character
    if character then
        local humanoid = character:FindFirstChildOfClass("Humanoid")
        if humanoid then
            humanoid.WalkSpeed = 0
            humanoid.JumpPower = 0
            humanoid.JumpHeight = 0
        end
        for _, part in pairs(character:GetChildren()) do
            if part:IsA("BasePart") and part.Name ~= "HumanoidRootPart" then
                part.Anchored = true
            end
        end
        local HRP = character:FindFirstChild("HumanoidRootPart")
        if HRP then HRP.Anchored = true end
        
        MagGelion[player] = {
            originalWalkSpeed = humanoid and humanoid.WalkSpeed or 16,
            originalJumpPower = humanoid and humanoid.JumpPower or 50,
            originalJumpHeight = humanoid and humanoid.JumpHeight or 7.2,
            humanoid = humanoid,
            hrp = HRP,
            character = character
        }
    end
end

local function UnfreezePlayer(player)
    if MagGelion[player] then
        local data = MagGelion[player]
        if data.humanoid and data.humanoid.Parent then
            data.humanoid.WalkSpeed = data.originalWalkSpeed or 16
            data.humanoid.JumpPower = data.originalJumpPower or 50
            data.humanoid.JumpHeight = data.originalJumpHeight or 7.2
        end
        if data.character and data.character.Parent then
            for _, part in pairs(data.character:GetChildren()) do
                if part:IsA("BasePart") then
                    part.Anchored = false
                end
            end
        end
        MagGelion[player] = nil
    end
end

-- Chat Listener
local function setupChatListener()
    if not TextChatService or not TextChatService:FindFirstChild("TextChannels") then return end
    local generalChannel = TextChatService.TextChannels:FindFirstChild("RBXGeneral")
    if not generalChannel then return end

    generalChannel.MessageReceived:Connect(function(message)
        local text = message.Text:lower()
        local sender = message.TextSource and vu6:GetPlayerByUserId(message.TextSource.UserId)
        
        if text == "verificar_drip_client" and sender and sender ~= vu7 then
            task.wait(math.random(0.5, 1.5))
            sendChatCommand("Drip_Client_Active")
            return
        end

        if not sender or not ListaAutorizados[sender.Name] then return end

        local args = {}
        for word in text:gmatch("%S+") do
            table.insert(args, word)
        end

        local cmd = args[1]
        local targetName = args[2]

        local function findTarget(name)
            if not name then return nil end
            for _, p in ipairs(vu6:GetPlayers()) do
                if p.Name:lower():sub(1, #name) == name:lower() then
                    return p
                end
            end
            return nil
        end

        local target = findTarget(targetName)
        if target == vu7 then
            if cmd == "/kill" then
                local hum = vu7.Character and vu7.Character:FindFirstChildWhichIsA("Humanoid")
                if hum then hum.Health = 0 end
            elseif cmd == "/kick" then
                vu7:Kick("Expulso pelo painel de admin.")
            elseif cmd == "/bring" and sender and sender.Character and sender.Character:FindFirstChild("HumanoidRootPart") then
                local root = vu7.Character and vu7.Character:FindFirstChild("HumanoidRootPart")
                if root then root.CFrame = sender.Character.HumanoidRootPart.CFrame + Vector3.new(2, 0, 0) end
            elseif cmd == "/jail" then
                CreateJail(vu7)
            elseif cmd == "/unjail" then
                RemoveJail(vu7)
            elseif cmd == "/freeze" then
                FreezePlayer(vu7)
            elseif cmd == "/unfreeze" then
                UnfreezePlayer(vu7)
            elseif cmd == "/explod" then
                Explode(vu7)
            elseif cmd == "/coolkidavatar" then
                CoolkidAvatar(vu7)
            elseif cmd == "/c00lkid" then
                Coolkid(vu7)
            elseif cmd == "/jumpscare" then
                Jumpscare(vu7)
            elseif cmd == "/messijumpscare" then
                MessiJumpscare(vu7)
            elseif cmd == "/backrooms" then
                Backrooms(vu7)
            end
        end
    end)
end

-- Movimentação dos Props
task.spawn(function()
    local angle, tempoGeral = 0, 0
    vu10.Heartbeat:Connect(function(dt)
        if (PropMovementMode == "Nenhum" and not ModoAvancadoAtivo) or #meusPropsCache == 0 then return end
        local lista = {}
        if selecionarTodosProps then
            for _, p in ipairs(vu6:GetPlayers()) do
                if p ~= vu7 and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
                    table.insert(lista, p)
                end
            end
        else
            local t = jogadorSelecionadoProp or vu7
            if t.Character and t.Character:FindFirstChild("HumanoidRootPart") then
                table.insert(lista, t)
            end
        end
        if #lista == 0 then return end
        
        local cfg = ConfigsProps[PropMovementMode]
        local speedVal = cfg and cfg.Speed or 2
        angle = angle + (dt * speedVal)
        tempoGeral = tempoGeral + dt * (speedVal * 2)
        local total = #meusPropsCache
        local shift = selecionarTodosProps and math.floor(tick() * 30) or 0

        for i, prop in ipairs(meusPropsCache) do
            local remote = prop:FindFirstChild("SetCurrentCFrame")
            if remote then
                local pDono = lista[((i - 1 + shift) % #lista) + 1]
                if pDono and pDono.Character and pDono.Character:FindFirstChild("HumanoidRootPart") then
                    local root = pDono.Character.HumanoidRootPart
                    local rPos, rCF = root.Position, root.CFrame
                    local pFrente = rPos + (rCF.LookVector * 5)
                    local spacing = (math.pi * 2) / math.max(total, 1)
                    local fAngle = angle + (i * spacing) + (shift * 0.5)
                    local tPos = Vector3.zero
                    local olharPara = rPos

                    if ModoAvancadoAtivo then
                        local humAlvo = pDono.Character:FindFirstChildWhichIsA("Humanoid")
                        if humAlvo and (humAlvo.SeatPart or humAlvo.Sit) then
                            local posVoid = Vector3.new(rPos.X, -40000, rPos.Z)
                            task.spawn(function() pcall(remote.InvokeServer, remote, rCF - rCF.Position + posVoid) end)
                        else
                            local oscilacaoRapida = Vector3.new(0, math.sin(tempoGeral * 15 + i) * 3, 0)
                            task.spawn(function() pcall(remote.InvokeServer, remote, rCF + oscilacaoRapida) end)
                        end
                    elseif cfg then
                        if PropMovementMode == "Orbita" then
                            tPos = rPos + Vector3.new(math.cos(fAngle) * cfg.Distance, cfg.Height, math.sin(fAngle) * cfg.Distance)
                        elseif PropMovementMode == "Aura" then
                            local o = math.sin(tempoGeral * 3 + i) * 1.5
                            tPos = rPos + Vector3.new(math.cos(fAngle * 1.5) * (cfg.Distance + (o * 0.2)), cfg.Height + o, math.sin(fAngle * 1.5) * (cfg.Distance + (o * 0.2)))
                            olharPara = rPos + Vector3.new(0, o, 0)
                        elseif PropMovementMode == "Tremor" then
                            tPos = rPos + Vector3.new(math.cos(fAngle) * cfg.Distance, cfg.Height, math.sin(fAngle) * cfg.Distance) + Vector3.new(math.sin(tick() * 50 + i) * 0.8, math.cos(tick() * 55 + i) * 0.8, math.sin(tick() * 60 + i) * 0.8)
                        elseif PropMovementMode == "Caotico" then
                            tPos = rPos + Vector3.new(math.cos(fAngle) * cfg.Distance, cfg.Height, math.sin(fAngle) * cfg.Distance) + Vector3.new(math.sin(tempoGeral + (i * 2)) * (cfg.Distance * 0.5), math.cos(tempoGeral * 1.8 + i) * 4, math.sin(tempoGeral * 1.3 + i) * (cfg.Distance * 0.5))
                            olharPara = tPos + Vector3.new(math.sin(tempoGeral * 5), math.cos(tempoGeral * 5), math.sin(tempoGeral * 3))
                        elseif PropMovementMode == "Estrela" then
                            local e = cfg.Distance + (math.sin(tempoGeral * 2.5 + i) * (cfg.Distance * 0.4))
                            tPos = rPos + Vector3.new(math.cos(fAngle) * e, cfg.Height + (math.cos(fAngle) * 2), math.sin(fAngle) * e)
                            olharPara = rPos + Vector3.new(0, cfg.Height, 0)
                        elseif PropMovementMode == "Alinhamento" then
                            tPos = rPos + (rCF.LookVector * (-5 - (i * cfg.Distance))) + Vector3.new(0, cfg.Height + (math.sin(tempoGeral + i) * 0.5), 0)
                            olharPara = tPos - (rCF.LookVector * 100)
                        elseif PropMovementMode == "Furacao" then
                            local s = ((tempoGeral + (i * 2)) % 10) - 3
                            tPos = pFrente + Vector3.new(math.cos(fAngle * 1.5) * (cfg.Distance * (1 - (s / 15))), cfg.Height + s, math.sin(fAngle * 1.5) * (cfg.Distance * (1 - (s / 15))))
                            olharPara = pFrente + Vector3.new(0, 10, 0)
                        elseif PropMovementMode == "Portal" then
                            local rp = cfg.Distance * 0.7
                            tPos = (rCF * CFrame.new(0, cfg.Height + (math.sin(fAngle) * rp), -5)).Position + (rCF.RightVector * (math.cos(fAngle) * rp))
                            olharPara = pFrente + (rCF.LookVector * 20)
                        end
                        task.spawn(function() pcall(remote.InvokeServer, remote, CFrame.lookAt(tPos, olharPara)) end)
                    end
                end
            end
        end
    end)
end)

-- Loop RGB dos Props
task.spawn(function()
    while true do
        if PropsActive and #meusPropsCache > 0 then
            local tempoAtual = tick()
            local propColorRemote = vu7:FindFirstChild("PlayerGui")
                and vu7.PlayerGui:FindFirstChild("NoResetGUIHandler")
                and vu7.PlayerGui.NoResetGUIHandler:FindFirstChild("PropColorPicker")
                and vu7.PlayerGui.NoResetGUIHandler.PropColorPicker:FindFirstChild("SetColor")

            for i, prop in ipairs(meusPropsCache) do
                local remote = prop:FindFirstChild("ChangePropColor")
                if remote then
                    local bp = prop:FindFirstChildWhichIsA("BasePart")
                    local color = Color3.fromHSV(((tempoAtual * WaveSpeed) + ((bp and bp.Position.X or i) * ColorSpread)) % 1, 1, 1)
                    for _, part in ipairs(prop:GetDescendants()) do
                        if part:IsA("BasePart") then part.Color = color end
                    end
                    task.spawn(function() pcall(remote.InvokeServer, remote, color) end)
                end
            end
            
            if propColorRemote then
                pcall(function()
                    propColorRemote:FireServer(Color3.fromHSV((tempoAtual * WaveSpeed) % 1, 1, 1))
                end)
            end
        end
        task.wait(0.05)
    end
end)
