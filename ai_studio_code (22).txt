    pcall(function()
        ball.CanCollide = false
        ball.Massless = true
        ball.CustomPhysicalProperties = PhysicalProperties.new(10000, 100, 100, 100, 100)

        local bv = ball:FindFirstChild("FlingPower") or Instance.new("BodyVelocity", ball)
        bv.Name = "FlingPower"
        bv.Velocity = Vector3.new(9e9, 9e9, 9e9)
        bv.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
        bv.P = 9e900

        bav.Name = "FlingSpin"
        bav.AngularVelocity = Vector3.new(9e9, 9e9, 9e9)

        ball.AssemblyLinearVelocity = Vector3.new(9e9, 9e9, 9e9)
        ball.AssemblyAngularVelocity = Vector3.new(9e9, 9e9, 9e9)
        ball:SetNetworkOwner(nil)

end

local function LimparFisicaFling(ball)
    if not ball or not ball:IsA("BasePart") then return end
    pcall(function()
        local bv = ball:FindFirstChild("FlingPower")
        if bv then bv:Destroy() end
        local bav = ball:FindFirstChild("FlingSpin")
        if bav then bav:Destroy() end
        ball.AssemblyLinearVelocity = Vector3.zero
        ball.AssemblyAngularVelocity = Vector3.zero
    end)
end

        end
    end)
    
    bringLoop = vu10.Heartbeat:Connect(function()
        if not running or not targetPlayer.Character or not tHRP or not myHRP then

            if isMoving then

                finalPos = frontCFrame.Position

        end)

        if tHum.Sit or (tick() - startTime > 1) then
            running = false
            bringLoop:Disconnect()
            noCollide:Disconnect()

            pcall(function()
                myHRP.Anchored = true
                myHRP.Velocity = Vector3.zero

                end

        end
